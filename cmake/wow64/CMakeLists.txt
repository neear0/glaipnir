cmake_minimum_required(VERSION 3.24)
project(glaipnir_wow64 LANGUAGES CXX)
set(CMAKE_VS_GLOBALS "VcpkgEnabled=false")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "WOW64 dump helper must be built for Win32")
endif()
add_executable(glaipnir_dump32 ../../glAIpnir/src/research/dump32_main.cpp)
set_target_properties(glaipnir_dump32 PROPERTIES OUTPUT_NAME glAIpnir-dump32)
target_include_directories(glaipnir_dump32 PRIVATE ../../glAIpnir/include)
target_compile_features(glaipnir_dump32 PRIVATE cxx_std_20)
target_compile_definitions(glaipnir_dump32 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX HARNESS_VERSION="${HARNESS_VERSION}")
target_compile_options(glaipnir_dump32 PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(glaipnir_dump32 PRIVATE dbghelp)
